# 触发词按钮注入技术文档

## 概述

本文档描述如何在 ComfyUI 节点的多行文本输入框左下角注入"选择触发词"按钮（TW 按钮）。该功能通过前端扩展脚本 `trigger_word_picker.js` 实现，支持 CLIPTextEncode、CR Text、LoRALoaderModelOnly、LoRALoaderFull、PrimitiveStringMultiline 等节点类型。

---

## 完整实现代码

以下是 `trigger_word_picker.js` 的完整源代码：

```javascript
import { app } from "../../../scripts/app.js";
import { saveTriggerWord, getAllTriggerWords } from "./utils/trigger-word-api.js";
import { showPreviewManager } from "./model_preview_manager.js";

/**
 * 触发词选择器 — 为 CLIPTextEncode 节点添加触发词快捷按钮
 *
 * 支持功能：
 * - 识别三种 LoRA 节点：LoRALoaderModelOnly（自定义）、LoraLoader（官方）、LoraLoaderModelOnly（官方仅模型）
 * - 自动比对工作流中的 LoRA 与配置文件
 * - 已保存的触发词：点击直接应用
 * - 未保存的触发词：提供输入框和保存按钮
 */

// LoRA 节点类型列表（自定义 + 官方）
const LORA_NODE_TYPES = ["LoRALoaderModelOnly", "LoRALoaderFull", "LoraLoader", "LoraLoaderModelOnly"];

let triggerWordsCache = [];
let lastFetchTime = 0;
const CACHE_TTL = 5000;
let currentButtonNode = null; // 当前点击按钮所属的节点

async function fetchAllTriggerWords() {
    const now = Date.now();
    if (now - lastFetchTime < CACHE_TTL && triggerWordsCache.length > 0) return triggerWordsCache;
    const words = await getAllTriggerWords();
    triggerWordsCache = words;
    lastFetchTime = now;
    return triggerWordsCache;
}

/**
 * 获取工作流中所有 LoRA 节点加载的 LoRA 文件名
 * 支持三种节点类型：LoRALoaderModelOnly（自定义）、LoraLoader（官方）、LoraLoaderModelOnly（官方仅模型）
 */
function getLoadedLoraNames() {
    const names = new Set();
    if (!app.graph) return names;
    for (const n of app.graph._nodes) {
        if (LORA_NODE_TYPES.includes(n.type)) {
            const w = n.widgets?.find(w => w.name === "lora_name");
            if (w?.value) names.add(w.value);
        }
    }
    return names;
}

/**
 * 保存触发词（弹窗内调用）
 */
async function saveTriggerWordInPopup(loraName, triggerWord) {
    const result = await saveTriggerWord(loraName, triggerWord);

    if (result.success) {
        // 清除缓存，下次打开时重新获取
        triggerWordsCache = [];
        lastFetchTime = 0;
    }

    return result;
}

/**
 * 同步触发词到工作流中对应的 LoRA 节点
 * @param {string} loraName - LoRA 文件名
 * @param {string} triggerWord - 触发词内容
 */
function syncTriggerWordToNode(loraName, triggerWord) {
    if (!app.graph) return;
    
    for (const node of app.graph._nodes) {
        if (!LORA_NODE_TYPES.includes(node.type)) continue;
        
        const loraWidget = node.widgets?.find(w => w.name === "lora_name");
        if (loraWidget?.value !== loraName) continue;
        
        const twWidget = node.widgets?.find(w => w.name === "trigger_word");
        if (twWidget) {
            twWidget.value = triggerWord;
            twWidget.callback?.(triggerWord);
            console.log(`[Comfyui-txtnode] 已同步触发词到节点 ${node.id}: ${loraName}`);
        }
    }
}

/**
 * 获取 LoRA 文件名的简短形式
 */
function getShortName(name) {
    return name.includes("\\") ? name.split("\\").pop() : name.includes("/") ? name.split("/").pop() : name;
}

/**
 * 应用触发词到对应的节点
 * @param {string} triggerWord - 触发词内容
 * @param {string} sourceLoraName - 触发词来源的 LoRA 名称（用于 LoRA 节点校验）
 */
function applyTriggerWord(triggerWord, sourceLoraName) {
    if (!currentButtonNode) {
        console.warn("[Comfyui-txtnode] 未找到当前节点");
        return;
    }

    // 如果当前节点是 LoRA 节点，检查 LoRA 名称是否匹配
    const LORA_BUTTON_TYPES = ["LoRALoaderModelOnly", "LoRALoaderFull"];
    if (LORA_BUTTON_TYPES.includes(currentButtonNode.type) && sourceLoraName) {
        const loraWidget = currentButtonNode.widgets?.find(w => w.name === "lora_name");
        const currentLoraName = loraWidget?.value;
        
        if (currentLoraName && currentLoraName !== sourceLoraName) {
            // LoRA 名称不匹配，弹出确认对话框
            const currentShort = getShortName(currentLoraName);
            const sourceShort = getShortName(sourceLoraName);
            
            if (!confirm(`当前节点加载的 LoRA 与触发词来源不匹配！\n\n当前节点: ${currentShort}\n触发词来源: ${sourceShort}\n\n是否仍要应用触发词？`)) {
                return; // 用户取消
            }
        }
    }

    // 支持 CLIPTextEncode/CR Text 的 "text" 字段、LoRA 节点的 "trigger_word" 字段、PrimitiveStringMultiline 的 "value" 字段
    const tw = currentButtonNode.widgets?.find(w => w.name === "text" || w.name === "trigger_word" || w.name === "value");
    if (tw) {
        tw.value = ((tw.value || "").trim() ? tw.value + ", " : "") + triggerWord;
        // 触发 widget 的回调以更新 UI
        tw.callback?.(tw.value);
        console.log("[Comfyui-txtnode] 已应用触发词到节点", currentButtonNode.id);
    } else {
        console.warn("[Comfyui-txtnode] 未找到文本输入框");
    }
}

function esc(s) { const d = document.createElement("div"); d.textContent = s; return d.innerHTML; }

/** 找到 textarea 的挂载容器 */
function getMountContainer(node) {
    // 支持 CLIPTextEncode/CR Text 的 "text" 字段、LoRA 节点的 "trigger_word" 字段、PrimitiveStringMultiline 的 "value" 字段
    const tw = node.widgets?.find(w => w.name === "text" || w.name === "trigger_word" || w.name === "value");
    if (!tw) return null;

    const el = tw.element || tw.inputEl;
    if (el && el.tagName === "TEXTAREA") {
        if (document.body.contains(el)) {
            return { textarea: el, parent: el.parentElement };
        }
    }

    const nc = document.querySelector(`[data-node-id="${node.id}"]`);
    if (nc) {
        const ta = nc.querySelector("textarea.p-textarea") || nc.querySelector("textarea");
        if (ta) return { textarea: ta, parent: ta.parentElement };
    }

    return null;
}

/** 注入按钮 */
function injectButton(node) {
    if (node._twBtn) return;
    node._twBtn = true;

    let attempts = 0;
    const poll = setInterval(() => {
        attempts++;
        const mc = getMountContainer(node);
        if (mc && mc.parent && !mc.parent.querySelector(".tw-btn")) {
            clearInterval(poll);

            if (getComputedStyle(mc.parent).position === "static") {
                mc.parent.style.position = "relative";
            }

            ["overflow", "overflowX", "overflowY"].forEach(p => {
                if (getComputedStyle(mc.parent)[p] === "hidden") {
                    mc.parent.style[p] = "visible";
                }
            });

            const btn = document.createElement("div");
            btn.className = "tw-btn";
            btn.title = "左键选择触发词，右键修改缩略图";
            btn.style.cssText = "position:absolute;left:7px;bottom:7px;z-index:999;" +
                "background:rgba(58,58,62,0.45);border:1px solid rgba(102,102,102,0.45);border-radius:3px;" +
                "cursor:pointer;width:20px;height:20px;display:flex;align-items:center;justify-content:center;" +
                "pointer-events:auto;user-select:none;";
            btn.onmouseenter = () => { btn.style.background = "rgba(74,106,176,0.6)"; btn.style.borderColor = "rgba(102,170,255,0.6)"; };
            btn.onmouseleave = () => { btn.style.background = "rgba(58,58,62,0.45)"; btn.style.borderColor = "rgba(102,102,102,0.45)"; };
            btn.onmousedown = (e) => e.stopPropagation();
            btn.onclick = (e) => {
                e.stopPropagation();
                currentButtonNode = node; // 记录当前节点
                showPopup();
            };
            // 右键：打开模型预览图管理弹窗
            btn.oncontextmenu = (e) => {
                e.preventDefault();
                e.stopPropagation();
                showPreviewManager(e);
            };

            const img = document.createElement("img");
            img.src = new URL("icon.png", import.meta.url).href;
            img.style.cssText = "width:16px;height:16px;pointer-events:none;";
            img.onerror = () => {
                btn.textContent = "TW";
                Object.assign(btn.style, { fontSize:"10px", fontWeight:"700", color:"#ddd", fontFamily:"Arial,sans-serif", lineHeight:"1", padding:"2px 4px" });
            };
            btn.appendChild(img);

            mc.parent.appendChild(btn);
            console.log("[Comfyui-txtnode] TW 按钮已挂载", node.id);

            const observer = new MutationObserver(() => {
                if (!document.body.contains(btn)) {
                    if (!mc.parent.querySelector(".tw-btn")) {
                        mc.parent.appendChild(btn);
                        console.log("[Comfyui-txtnode] TW 按钮被移除后恢复", node.id);
                    }
                }
            });
            observer.observe(mc.parent, { childList: true, subtree: true });
            return;
        }
        if (attempts > 80) clearInterval(poll);
    }, 100);
}

// 支持注入 TW 按钮的节点类型
const TW_BUTTON_NODE_TYPES = ["CLIPTextEncode", "CR Text", "LoRALoaderModelOnly", "LoRALoaderFull", "PrimitiveStringMultiline"];

/** 检查节点是否支持 TW 按钮 */
function isTWButtonSupported(node) {
    return TW_BUTTON_NODE_TYPES.includes(node?.type);
}

app.registerExtension({
    name: "Comfyui-txtnode.TriggerWordPicker",

    async setup() {
```

## 核心实现流程

整个实现分为四个关键步骤：

```
节点类型注册 → 查找 textarea 容器 → 注入按钮 DOM → 监听节点生命周期
```

---

## 1. 节点类型注册

通过常量数组声明需要注入按钮的节点类型：

```javascript
const TW_BUTTON_NODE_TYPES = [
    "CLIPTextEncode",
    "CR Text",
    "LoRALoaderModelOnly",
    "LoRALoaderFull",
    "PrimitiveStringMultiline"
];

function isTWButtonSupported(node) {
    return TW_BUTTON_NODE_TYPES.includes(node?.type);
}
```

**设计要点：**
- 新增节点类型只需在数组中添加对应字符串即可
- `PrimitiveStringMultiline` 是 ComfyUI 官方"字符串（多行）"节点的类型名

---

## 2. 查找 textarea 挂载容器

`getMountContainer(node)` 函数负责定位节点中多行文本输入框的 DOM 元素及其父容器：

```javascript
function getMountContainer(node) {
    // 第一步：通过 widget name 查找对应的输入控件
    const tw = node.widgets?.find(w =>
        w.name === "text" || w.name === "trigger_word" || w.name === "value"
    );
    if (!tw) return null;

    // 第二步：优先通过 widget 的 element/inputEl 属性获取 textarea
    const el = tw.element || tw.inputEl;
    if (el && el.tagName === "TEXTAREA") {
        if (document.body.contains(el)) {
            return { textarea: el, parent: el.parentElement };
        }
    }

    // 第三步：降级方案 —— 通过 data-node-id 属性在 DOM 树中搜索
    const nc = document.querySelector(`[data-node-id="${node.id}"]`);
    if (nc) {
        const ta = nc.querySelector("textarea.p-textarea") || nc.querySelector("textarea");
        if (ta) return { textarea: ta, parent: ta.parentElement };
    }

    return null;
}
```

**三层查找策略：**

| 层级 | 方法 | 适用场景 |
|------|------|---------|
| 1 | `widget.element` / `widget.inputEl` | 控件已渲染且挂载到 DOM |
| 2 | `document.body.contains(el)` 验证 | 确保元素仍在文档中 |
| 3 | `[data-node-id]` DOM 查询 | 控件属性不可用时的降级方案 |

**widget name 映射关系：**

| 节点类型 | widget name |
|---------|------------|
| CLIPTextEncode | `text` |
| CR Text | `text` |
| LoRALoaderModelOnly / LoRALoaderFull | `trigger_word` |
| PrimitiveStringMultiline | `value` |

## 3. 注入按钮 DOM

`injectButton(node)` 函数将按钮元素挂载到 textarea 父容器的左下角：

```javascript
function injectButton(node) {
    // 防止重复注入
    if (node._twBtn) return;
    node._twBtn = true;

    let attempts = 0;
    const poll = setInterval(() => {
        attempts++;
        const mc = getMountContainer(node);
        if (mc && mc.parent && !mc.parent.querySelector(".tw-btn")) {
            clearInterval(poll);

            // 3.1 设置父容器为相对定位（按钮绝对定位的参照物）
            if (getComputedStyle(mc.parent).position === "static") {
                mc.parent.style.position = "relative";
            }

            // 3.2 解除 overflow:hidden 限制（否则按钮会被裁剪）
            ["overflow", "overflowX", "overflowY"].forEach(p => {
                if (getComputedStyle(mc.parent)[p] === "hidden") {
                    mc.parent.style[p] = "visible";
                }
            });

            // 3.3 创建按钮元素
            const btn = document.createElement("div");
            btn.className = "tw-btn";
            btn.style.cssText = "position:absolute;left:7px;bottom:7px;z-index:999;" +
                "background:rgba(58,58,62,0.45);border:1px solid rgba(102,102,102,0.45);" +
                "border-radius:3px;cursor:pointer;width:20px;height:20px;" +
                "display:flex;align-items:center;justify-content:center;" +
                "pointer-events:auto;user-select:none;";

            // 3.4 绑定事件
            btn.onclick = (e) => {
                e.stopPropagation();
                currentButtonNode = node;
                showPopup();
            };
            btn.oncontextmenu = (e) => {
                e.preventDefault();
                e.stopPropagation();
                showPreviewManager(e);
            };

            // 3.5 挂载按钮
            mc.parent.appendChild(btn);

            // 3.6 MutationObserver 监听按钮被移除后自动恢复
            const observer = new MutationObserver(() => {
                if (!document.body.contains(btn)) {
                    if (!mc.parent.querySelector(".tw-btn")) {
                        mc.parent.appendChild(btn);
                    }
                }
            });
            observer.observe(mc.parent, { childList: true, subtree: true });
            return;
        }
        if (attempts > 80) clearInterval(poll);
    }, 100);
}
```

### 关键技术点

#### 3.1 定位策略

```
父容器: position: relative  ← 作为定位上下文
按钮:   position: absolute; left: 7px; bottom: 7px  ← 左下角定位
```

#### 3.2 轮询机制

由于 ComfyUI 节点渲染是异步的，textarea 可能在节点创建后延迟出现。采用轮询方式等待 DOM 就绪：

- 每 100ms 检查一次
- 最多尝试 80 次（8 秒超时）
- 找到容器后立即清除定时器

#### 3.3 防重复注入

通过 `node._twBtn` 标记位防止同一节点被多次注入按钮。

#### 3.4 按钮持久化

使用 `MutationObserver` 监听父容器的 DOM 变化。当 ComfyUI 重新渲染节点导致按钮被移除时，自动重新挂载。

#### 3.5 事件处理

| 事件 | 行为 |
|------|------|
| `onclick` | 记录当前节点引用，弹出触发词选择弹窗 |
| `oncontextmenu` | 弹出模型预览图管理弹窗 |
| `onmousedown` | `stopPropagation()` 防止触发节点拖拽 |

## 4. 节点生命周期监听

通过 ComfyUI 扩展 API 在两个时机注入按钮：

```javascript
app.registerExtension({
    name: "Comfyui-txtnode.TriggerWordPicker",

    async setup() {
        // 4.1 等待 graph 初始化
        const wait = setInterval(() => {
            if (!app.graph) return;
            clearInterval(wait);

            // 4.2 为已存在的节点注入按钮（延迟多次尝试）
            [500, 2000, 5000].forEach(d => setTimeout(() => {
                for (const n of app.graph._nodes) {
                    if (isTWButtonSupported(n)) injectButton(n);
                }
            }, d));

            // 4.3 监听后续新增的节点
            if (!app.graph._twHooked) {
                app.graph._twHooked = true;
                const orig = app.graph.onNodeAdded;
                app.graph.onNodeAdded = function (node) {
                    orig?.call(this, node);
                    if (isTWButtonSupported(node)) {
                        [300, 1500, 4000].forEach(d =>
                            setTimeout(() => injectButton(node), d)
                        );
                    }
                };
            }
        }, 100);
    },
});
```

### 两个注入时机

| 时机 | 方法 | 说明 |
|------|------|------|
| 页面加载时 | 遍历 `app.graph._nodes` | 处理工作流中已存在的节点 |
| 运行时新增 | 重写 `onNodeAdded` 回调 | 处理用户拖入的新节点 |

### 多次延迟策略

对每个节点在多个时间点（300ms、1500ms、4000ms）尝试注入，确保即使节点渲染较慢也能成功挂载按钮。`injectButton` 内部的 `_twBtn` 标记保证不会重复注入。

## 5. 触发词应用逻辑

点击按钮弹出选择弹窗后，用户选择一个触发词，通过 `applyTriggerWord()` 写入对应节点的文本框：

```javascript
function applyTriggerWord(triggerWord, sourceLoraName) {
    if (!currentButtonNode) return;

    // 查找目标 widget（text / trigger_word / value）
    const tw = currentButtonNode.widgets?.find(w =>
        w.name === "text" || w.name === "trigger_word" || w.name === "value"
    );
    if (tw) {
        // 追加到现有内容末尾
        tw.value = ((tw.value || "").trim() ? tw.value + ", " : "") + triggerWord;
        tw.callback?.(tw.value);  // 触发 UI 更新
    }
}
```

**追加逻辑：**
- 文本框为空 → 直接写入触发词
- 文本框已有内容 → 用 `, ` 分隔后追加

## 6. 扩展新节点类型的步骤

若需要为其他节点类型添加 TW 按钮，只需两步：

**步骤 1：** 在 `TW_BUTTON_NODE_TYPES` 数组中添加节点类型名

```javascript
const TW_BUTTON_NODE_TYPES = [
```

## 7. 注意事项

1. **CSS 定位依赖**：按钮使用 `position: absolute` 相对于 textarea 父容器定位，必须确保父容器为 `position: relative`
2. **overflow 问题**：ComfyUI 某些节点的容器可能设置 `overflow: hidden`，会导致按钮被裁剪，代码中已自动修正为 `visible`
3. **节点渲染时序**：ComfyUI 节点 DOM 是异步渲染的，必须使用轮询 + 多次延迟策略确保注入时机正确
4. **MutationObserver 开销**：每个注入按钮的节点都会创建一个 Observer，在大量节点场景下需注意性能
5. **widget name 差异**：不同节点类型的文本输入 widget name 不同，必须逐一确认
